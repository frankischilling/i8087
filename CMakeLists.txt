cmake_minimum_required(VERSION 3.16)
project(i8087 VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(I8087_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(I8087_BUILD_TESTS "Build tests" ON)
add_library(i8087 STATIC src/i8087.c src/soft80.c src/transcend.c)
target_include_directories(i8087 PUBLIC include PRIVATE src)
if(MSVC)
  target_compile_options(i8087 PRIVATE /W4)
else()
  target_compile_options(i8087 PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(I8087_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(i8087 PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_executable(i8087_replay tools/replay.c)
target_link_libraries(i8087_replay PRIVATE i8087)
add_executable(i8087_demo examples/demo.c)
target_link_libraries(i8087_demo PRIVATE i8087)
if(I8087_BUILD_TESTS)
  enable_testing()
  add_executable(i8087_tests tests/test_i8087.c)
  target_include_directories(i8087_tests PRIVATE src)
  target_link_libraries(i8087_tests PRIVATE i8087)
  add_test(NAME instructions COMMAND i8087_tests)
  add_executable(i8087_regressions tests/test_regressions.c)
  target_include_directories(i8087_regressions PRIVATE src)
  target_link_libraries(i8087_regressions PRIVATE i8087)
  add_test(NAME original_8087_regressions COMMAND i8087_regressions)
  add_executable(i8087_stress tests/test_stress.c)
  target_link_libraries(i8087_stress PRIVATE i8087)
  add_test(NAME raw_state_safety COMMAND i8087_stress)
  add_executable(i8087_numeric_cli tests/numeric_cli.c)
  target_include_directories(i8087_numeric_cli PRIVATE src)
  target_link_libraries(i8087_numeric_cli PRIVATE i8087)
  find_package(Python3 3.10 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME rational_oracle COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_fraction.py $<TARGET_FILE:i8087_numeric_cli>)
    add_test(NAME instruction_fraction_oracle COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_instruction_fraction.py $<TARGET_FILE:i8087_replay>)
    add_test(NAME trace_tool_contracts COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_trace_tools.py $<TARGET_FILE:i8087_replay>)
  endif()
endif()

# Keep test/example diagnostics as strict as the library's diagnostics.
foreach(program i8087_replay i8087_demo i8087_tests i8087_regressions i8087_stress i8087_numeric_cli)
  if(TARGET ${program})
    if(MSVC)
      target_compile_options(${program} PRIVATE /W4)
    else()
      target_compile_options(${program} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
  endif()
endforeach()
if(I8087_BUILD_TESTS)
  set_tests_properties(instructions original_8087_regressions raw_state_safety PROPERTIES TIMEOUT 120)
  if(Python3_Interpreter_FOUND)
    set_tests_properties(rational_oracle instruction_fraction_oracle trace_tool_contracts PROPERTIES TIMEOUT 120)
  endif()
endif()
